ttf_summary <- function(df, group_var) {
  df %>%
    filter(!is.na(.data[[group_var]])) %>%
    group_by(.data[[group_var]]) %>%
    summarize(
      mean_ttf   = mean(time_to_file, na.rm = TRUE),
      median_ttf = median(time_to_file, na.rm = TRUE),
      se         = sd(time_to_file, na.rm = TRUE) / sqrt(n()),
      ci_lower   = mean_ttf - 1.96 * se,
      ci_upper   = mean_ttf + 1.96 * se
    )
}

ttf_panel <- function(group_var, xlab) {
  ggplot(ttf_summary(isaac_df, group_var), aes(x = .data[[group_var]])) +
    geom_pointrange(aes(y = mean_ttf, ymin = ci_lower, ymax = ci_upper, color = "Mean"),
                    size = 0.6) +
    geom_point(aes(y = median_ttf, color = "Median"), shape = 17, size = 2) +
    scale_color_manual(values = c("Mean" = "steelblue", "Median" = "steelblue4"),
                       name = NULL) +
    labs(x = xlab, y = "Time to File (Years)") +
    theme_bw() +
    theme(legend.position = "bottom")
}

fig <- arrangeGrob(
  ttf_panel("rule_of_law_3cats", "Rule of Law"),
  ttf_panel("income_group", "Respondent State GDP") + theme(axis.title.y = element_blank()),
  ttf_panel("case_outcome", "Case Outcome"),
  ttf_panel("indir_exp_factor", "Claim Type") + theme(axis.title.y = element_blank()),
  nrow = 2, ncol = 2
)

prop_summary_dom <- function(df, group_var) {
  df %>%
    filter(!is.na(domestic_suit), !is.na(.data[[group_var]])) %>%
    group_by(.data[[group_var]]) %>%
    summarize(
      n        = n(),
      prop_dom = mean(domestic_suit, na.rm = TRUE),
      se       = sqrt(prop_dom * (1 - prop_dom) / n),
      ci_lower = pmax(0, prop_dom - 1.96 * se),
      ci_upper = pmin(1, prop_dom + 1.96 * se),
      .groups  = "drop"
    )
}

}

type_labels_lookup <- c(total_inline       = "Inline",
                        total_partial_line = "Partial line",
                        total_full_line    = "Full line",
                        total_block        = "Block")
type_levels <- c("Inline", "Partial line", "Full line", "Block")

ts_types <- redactions_df %>%
  filter(!is.na(year)) %>%
  select(year, total_inline, total_partial_line, total_full_line, total_block) %>%
  pivot_longer(-year, names_to = "type", values_to = "count") %>%
  mutate(type = factor(type_labels_lookup[type], levels = type_levels)) %>%
  group_by(year, type) %>%
  summarise(total = sum(count), .groups = "drop") %>%
  group_by(year) %>%
  filter(sum(total) > 0) %>%
  ungroup()

label_df <- ts_types %>%
  filter(year == max(year)) %>%
  arrange(desc(type)) %>%
  mutate(share = total / sum(total),
         ypos  = cumsum(share) - share / 2)

fig <- ggplot(ts_types, aes(year, total, fill = type)) +
  geom_area(position = "fill", alpha = 0.85, color = "white", linewidth = 0.2) +
  geom_text(data = label_df,
            aes(x = year, y = ypos, label = type),
            hjust = 0, nudge_x = 0.3, size = 3.5, color = "black") +
  scale_fill_grey(start = 0.8, end = 0.3) +
  scale_y_continuous(labels = scales::label_percent()) +
  scale_x_continuous(expand = expansion(mult = c(0.02, 0.18))) +
  labs(x = NULL, y = "Share of Redactions", fill = NULL) +
  theme_bw() +
  theme(legend.position = "none",
        panel.grid.major = element_blank(),
        panel.grid.minor = element_blank())

attrition_stages <- tribble(
  ~stage_idx, ~stage_short,
  1,          "Filed",
  2,          "Decided",
  3,          "Investor wins",
  4,          "≥ 50% of claim",
  5,          "≥ 75% of claim"
)

stage_filter <- function(d, idx) {
  if (idx == 1) return(d)
  if (idx == 2) return(d %>% filter(!outcome %in% c("Pending", "Data not available"),
                                    !is.na(outcome)))
  if (idx == 3) return(d %>% filter(outcome == "Decided in favour of investor"))
  if (idx == 4) return(d %>% filter(outcome == "Decided in favour of investor",
                                    !is.na(prop_award), prop_award >= 0.50))
  if (idx == 5) return(d %>% filter(outcome == "Decided in favour of investor",
                                    !is.na(prop_award), prop_award >= 0.75))
}

stage_summary <- function(d) {
  inv_win <- d %>% filter(outcome == "Decided in favour of investor")
  tibble(n_cases      = nrow(d),
         total_claims = sum(d$claims, na.rm = TRUE),
         total_awards = sum(inv_win$awards, na.rm = TRUE))
}

attrition <- attrition_stages %>%
  mutate(summary = map(stage_idx, ~ stage_summary(stage_filter(award_full, .x)))) %>%
  unnest(summary) %>%
  mutate(pct_cases = n_cases / first(n_cases))

case_max  <- max(attrition$n_cases)
claim_max <- max(attrition$total_claims)

base_layers <- list(
  geom_vline(xintercept = attrition$stage_idx, color = "grey92", linewidth = 0.3),
  geom_hline(yintercept = 0, color = "grey85", linewidth = 0.3)
)

frame_theme <- theme_minimal() +
  theme(axis.text.y  = element_blank(),
        panel.grid   = element_blank(),
        panel.border = element_rect(color = "grey85", fill = NA),
        plot.margin  = margin(5, 10, 5, 10))

stage_axis <- scale_x_continuous(breaks = attrition$stage_idx,
                                 labels = attrition$stage_short,
                                 expand = expansion(add = 0.4))

p_cases <- ggplot(attrition, aes(stage_idx)) +
  base_layers +
  geom_ribbon(aes(ymin = -n_cases / 2, ymax = n_cases / 2),
              fill = "#33373b", alpha = 0.9) +
  geom_text(aes(y = n_cases / 2, label = scales::comma(n_cases)),
            vjust = -0.6, size = 3.6, fontface = "bold") +
  geom_text(aes(y = -n_cases / 2,
                label = scales::percent(pct_cases, accuracy = 1)),
            vjust = 1.6, size = 3.2, color = "grey30") +
  stage_axis +
  coord_cartesian(ylim = c(-case_max * 0.7, case_max * 0.7)) +
  labs(x = NULL, y = NULL) +
  frame_theme

p_dollars <- ggplot(attrition, aes(stage_idx)) +

# Recent years have claims but no awards yet, so the series ends at the last year with an award proportion.
last_year <- max(award_full$year_init[!is.na(award_full$prop_award)], na.rm = TRUE)

award_prop_yearly <- award_full %>%
  group_by(year_init) %>%
  summarize(n = n(),
            mean_prop  = mean(prop_award, na.rm = TRUE),
            mean_claim = mean(claims, na.rm = TRUE),
            .groups = "drop") %>%
  filter(year_init >= 1992, year_init <= last_year)

scale_factor <- max(award_prop_yearly$mean_prop, na.rm = TRUE) /
  max(award_prop_yearly$mean_claim, na.rm = TRUE)

fig <- ggplot(award_prop_yearly, aes(x = year_init)) +
  geom_col(aes(y = n / max(n) * max(mean_prop, na.rm = TRUE) * 0.2),
           fill = "gray70", alpha = 0.4, width = 0.7) +
  geom_line(aes(y = mean_prop, color = "Mean Award Proportion"),
            linewidth = 0.9) +
  geom_line(aes(y = mean_claim * scale_factor, color = "Mean Claim Size"),
            linewidth = 0.9, linetype = "dashed") +
  scale_y_continuous(name = "Mean Award Proportion",
                     sec.axis = sec_axis(~ . / scale_factor,
                                         name = "Mean Claim Size (USD)")) +
  scale_color_manual(values = c("Mean Award Proportion" = "steelblue",
                                "Mean Claim Size" = "steelblue4"),
                     name = NULL) +
  scale_x_continuous(breaks = seq(1995, 2020, by = 5)) +
  labs(x = "Year") +
  theme_minimal() +
  theme(legend.position = "bottom")
